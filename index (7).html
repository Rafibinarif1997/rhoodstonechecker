const CONFIG={chainId:4663,chainName:'Robinhood Chain',rpcUrl:'https://rpc.mainnet.chain.robinhood.com',explorer:'https://robinhoodchain.blockscout.com',contract:'0x6be906e10351B4a970521c386E89D9e4e34c47C9'}; const ABI=['function balanceOf(address owner) view returns (uint256)']; const $=s=>document.querySelector(s); const short=a=>a?`${a.slice(0,6)}…${a.slice(-4)}`:'—'; const tier=n=>n>=8?'DIAMOND':n>=4?'GOLD':n>=2?'SILVER':'STONE'; function provider(){if(window.ethereum)return window.ethereum; throw Error('No wallet detected. Open this page in MetaMask, Rabby, OKX, or another EVM wallet browser.');} async function network(){const e=provider(), id='0x'+CONFIG.chainId.toString(16); try{await e.request({method:'wallet_switchEthereumChain',params:[{chainId:id}]})}catch(x){if(x.code!==4902)throw x; await e.request({method:'wallet_addEthereumChain',params:[{chainId:id,chainName:CONFIG.chainName,nativeCurrency:{name:'Ether',symbol:'ETH',decimals:18},rpcUrls:[CONFIG.rpcUrl],blockExplorerUrls:[CONFIG.explorer]}]})}} async function connect(){const e=provider(); await e.request({method:'eth_requestAccounts'}); await network(); const p=new ethers.BrowserProvider(e); const s=await p.getSigner(); const a=await s.getAddress(); const c=new ethers.Contract(CONFIG.contract,ABI,p); const b=Number(await c.balanceOf(a)); localStorage.setItem('rhood_wallet',a); localStorage.setItem('rhood_balance',b); return {address:a,balance:b,tier:tier(b)};} async function verifyExisting(){try{const e=provider(); const ac=await e.request({method:'eth_accounts'}); if(!ac?.length)return null; await network(); const p=new ethers.BrowserProvider(e), c=new ethers.Contract(CONFIG.contract,ABI,p); const b=Number(await c.balanceOf(ac[0])); localStorage.setItem('rhood_wallet',ac[0]); localStorage.setItem('rhood_balance',b); return {address:ac[0],balance:b,tier:tier(b)};}catch{return null}} function render(x){document.querySelectorAll('[data-address]').forEach(n=>n.textContent=short(x?.address));document.querySelectorAll('[data-balance]').forEach(n=>n.textContent=x?.balance??'—');document.querySelectorAll('[data-tier]').forEach(n=>n.textContent=x?.tier??'—');} async function boot(){document.querySelectorAll('[data-connect]').forEach(b=>b.addEventListener('click',async()=>{b.disabled=true;b.textContent='Connecting…';try{const x=await connect(); if(x.balance<1){alert('This wallet does not hold a RhoodStone NFT.');return} render(x); if(location.pathname!='/portal/'&&!location.pathname.endsWith('/portal')) location.href='/portal/'; else document.querySelector('#holder').classList.remove('hidden');}catch(e){alert(e.message||'Wallet connection failed.')}finally{b.disabled=false;b.textContent='Connect Wallet'}})); const x=await verifyExisting(); render(x); if(location.pathname.includes('/portal')&&x?.balance>0){$('#gate')?.classList.add('hidden');$('#holder')?.classList.remove('hidden')} if(window.ethereum){window.ethereum.on('accountsChanged',()=>location.reload());window.ethereum.on('chainChanged',()=>location.reload())}} boot();
